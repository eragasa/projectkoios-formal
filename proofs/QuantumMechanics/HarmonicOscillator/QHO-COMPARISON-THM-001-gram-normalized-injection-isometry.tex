\section{Proof schematic}
\begin{center}\fbox{\parbox{0.88\textwidth}{\small\raggedright $J=SR$ $\Rightarrow$ $J^{\mathsf T}J=R^{\mathsf T}(S^{\mathsf T}S)R$ $\Rightarrow$ supplied exact Gram identity $\Rightarrow$ $J^{\mathsf T}J=I$}}\end{center}

\section{Human-readable mathematical argument}
\paragraph{Proof status.} Complete for the exact Lean proposition.
\paragraph{Argument.} Expand the injection as $J=SR$.  Transposition reverses products, so $J^{\mathsf T}J=R^{\mathsf T}S^{\mathsf T}SR$.  Reassociate to obtain $R^{\mathsf T}(S^{\mathsf T}S)R$, then apply the supplied exact normalization identity.
